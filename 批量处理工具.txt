import os
import pandas as pd
from src.preprocess import TextPreprocessor
from src.classifier import ReviewClassifier
import time

class BatchProcessor:
    def __init__(self, preprocessor: TextPreprocessor, classifier: ReviewClassifier):
        self.preprocessor = preprocessor
        self.classifier = classifier

    def process_folder(self, folder_path: str, output_path: str):
        """批量处理文件夹内所有CSV评论文件"""
        all_results = []
        start_time = time.time()
        file_count = 0

        for filename in os.listdir(folder_path):
            if filename.endswith(".csv"):
                file_path = os.path.join(folder_path, filename)
                df = pd.read_csv(file_path)
                if "comment" not in df.columns:
                    continue
                
                # 批量预测
                df["pred_label"] = df["comment"].apply(
                    lambda x: self.classifier.predict_single(x, self.preprocessor)
                )
                all_results.append(df)
                file_count += 1
                print(f"[批量处理] 已完成 {filename}")

        # 合并输出
        final_df = pd.concat(all_results, ignore_index=True)
        final_df.to_csv(output_path, index=False, encoding="utf-8-sig")
        
        cost_time = time.time() - start_time
        print(f"\n[批量处理完成] 共处理 {file_count} 个文件，{len(final_df)} 条数据")
        print(f"耗时: {cost_time:.2f} 秒，单条平均耗时: {cost_time/len(final_df):.4f} 秒")
        print(f"结果已保存至 {output_path}")
